C.1 The Core Rule as a Prisma Comment

model Voter {
  id                      String  @id @default(uuid())
  voterNumber             String  @unique
  fullName                String
  // The registered constituency is the ONLY source of a ballot.
  // No remote polling workflow may write this field.
  registeredConstituencyId String
  registeredConstituency  Constituency @relation(fields: [registeredConstituencyId], references: [id])
  homePollingCentreId     String?
  homePollingCentre       PollingCentre? @relation(fields: [homePollingCentreId], references: [id])
  createdAt               DateTime @default(now())
}

model RemotePollingAssignment {
  id              String  @id @default(uuid())
  requestId       String  @unique
  voterId         String
  electionId      String
  pollingCentreId String
  assignedById    String
  createdAt       DateTime @default(now())
  // Deliberately NO constituencyId: an assignment changes where,
  // never what.
  @@unique([voterId, electionId])
}

model SubmittedBallot {
  ballotReference String   @id
  electionId      String
  constituencyId  String   // the voter's REGISTERED constituency
  nonce           Bytes
  ciphertext      Bytes
  authTag         Bytes
  submittedAt     DateTime @default(now())
  // Deliberately NO voterId and no relation to Voter or
  // VoterElectionStatus: the identity side and the choice side

}

C.2 The Transition Table

export const TRANSITIONS: Record<Status, readonly Status[]> = {
  PENDING:      ['UNDER_REVIEW', 'WITHDRAWN', 'EXPIRED'],
  UNDER_REVIEW: ['APPROVED', 'REJECTED', 'WITHDRAWN', 'PENDING', 'EXPIRED'],
  APPROVED:     ['USED', 'EXPIRED'],
  REJECTED:     [],
  WITHDRAWN:    [],
  USED:         [],
  EXPIRED:      [],
} as const;

export function assertLegal(from: Status, to: Status): void {
  if (!TRANSITIONS[from].includes(to))
    throw new IllegalTransitionError(from, to);   // → HTTP 409
}

C.3 The Verification and Counting Split

// Returns a boolean. Exposes no plaintext to any caller.
export function verifyBallot(sealed: Sealed): boolean {
  try {
    const decipher = crypto.createDecipheriv('aes-256-gcm', key, sealed.nonce);
    decipher.setAAD(Buffer.from(AAD(sealed.ref, sealed.electionId, sealed.constituencyId)));
    decipher.setAuthTag(sealed.authTag);
    decipher.update(sealed.ciphertext);
    decipher.final();
    return true;
  } catch { return false; }
}

// Returns the plaintext. Called only from the simulator, only after closure.
export function openBallotForCounting(sealed: Sealed): string {
  const decipher = crypto.createDecipheriv('aes-256-gcm', key, sealed.nonce);
  decipher.setAAD(Buffer.from(AAD(sealed.ref, sealed.electionId, sealed.constituencyId)));
  decipher.setAuthTag(sealed.authTag);
  return Buffer.concat([decipher.update(sealed.ciphertext), decipher.final()])
    .toString('utf8');
}

The split is the point. Two functions, two return types, one caller each. A path that must not disclose a choice is given the function that cannot.
